# 파이프라인 배포 — data → train → eval → (품질 게이트) → 모델 S3 등록 (KFP v2)

> 앞 가이드(KFP/KServe 설치)에 이어지는 단계.
> 모델은 scikit-learn(iris)라 GPU 없이 동작. 검증: KFP SDK 2.x

## 데이터 흐름
```
prepare_data ─train_out─▶ train ─model─▶ evaluate ─accuracy─▶ dsl.If(acc ≥ threshold)
      └────────test_out────────────────▶            (통과 시만) ─▶ register_model
                                                                   → s3://models/iris/v1/model.joblib
```

## 사전 준비 (3가지)
```bash
# 1) 로컬 SDK + k8s 확장
pip install "kfp>=2.7" kfp-kubernetes

# 2) S3(또는 MinIO) 버킷 1개  (예: models)

# 3) 파이프라인 실행 네임스페이스(kubeflow)에 S3 자격증명 secret 생성
kubectl create secret generic s3-creds -n kubeflow \
  --from-literal=AWS_ACCESS_KEY_ID=<YOUR_KEY> \
  --from-literal=AWS_SECRET_ACCESS_KEY=<YOUR_SECRET>
```
> 이 `s3-creds`는 **파이프라인이 도는 네임스페이스(kubeflow)** 에 있어야 함.
> (KServe가 읽을 secret은 **서빙 네임스페이스**에 따로 필요 — 다음 단계)

## pipeline.py
```python
from kfp import dsl
from kfp.dsl import Dataset, Model, Metrics, Input, Output
from kfp import kubernetes          # pip install kfp-kubernetes

# 모든 스텝 공통 베이스 이미지 + 런타임 설치 패키지 (커스텀 이미지 불필요)
COMMON = dict(
    base_image="python:3.11",
    packages_to_install=["scikit-learn==1.5.2", "pandas", "joblib"],
)

# ── ① 데이터 준비: iris 로드 → train/test 분할 → Dataset 아티팩트 출력 ──
@dsl.component(**COMMON)
def prepare_data(train_out: Output[Dataset], test_out: Output[Dataset]):
    import pandas as pd
    from sklearn.datasets import load_iris
    from sklearn.model_selection import train_test_split

    X, y = load_iris(return_X_y=True, as_frame=True)
    df = X.copy()
    df["target"] = y
    train_df, test_df = train_test_split(
        df, test_size=0.2, random_state=42, stratify=y
    )
    train_df.to_csv(train_out.path, index=False)
    test_df.to_csv(test_out.path, index=False)

# ── ② 학습: RandomForest 학습 → model.joblib 저장 ──
#     (KServe sklearn 런타임이 기대하는 파일명)
@dsl.component(**COMMON)
def train(train_data: Input[Dataset], model_out: Output[Model]):
    import os, joblib
    import pandas as pd
    from sklearn.ensemble import RandomForestClassifier

    df = pd.read_csv(train_data.path)
    X, y = df.drop(columns=["target"]), df["target"]
    clf = RandomForestClassifier(n_estimators=100, random_state=42)
    clf.fit(X, y)

    os.makedirs(model_out.path, exist_ok=True)
    joblib.dump(clf, os.path.join(model_out.path, "model.joblib"))

# ── ③ 평가: 정확도 계산 → Metrics 기록 + float 반환(게이트 판단용) ──
@dsl.component(**COMMON)
def evaluate(
    model_in: Input[Model],
    test_data: Input[Dataset],
    metrics: Output[Metrics],
) -> float:
    import os, joblib
    import pandas as pd
    from sklearn.metrics import accuracy_score

    clf = joblib.load(os.path.join(model_in.path, "model.joblib"))
    df = pd.read_csv(test_data.path)
    X, y = df.drop(columns=["target"]), df["target"]

    acc = float(accuracy_score(y, clf.predict(X)))
    metrics.log_metric("accuracy", acc)      # KFP UI에 표시
    return acc

# ── ④ 등록(승격): 통과한 모델만 S3의 안정적 경로로 업로드 ──
@dsl.component(base_image="python:3.11", packages_to_install=["boto3"])
def register_model(
    model_in: Input[Model],
    bucket: str,
    version: str,
    s3_endpoint: str = "",
):
    import os, boto3
    kwargs = {"endpoint_url": s3_endpoint} if s3_endpoint else {}
    s3 = boto3.client("s3", **kwargs)        # 자격증명은 secret→env로 주입됨
    key = f"iris/{version}/model.joblib"
    s3.upload_file(os.path.join(model_in.path, "model.joblib"), bucket, key)
    print(f"registered: s3://{bucket}/{key}")

# ── 파이프라인 정의 (DAG) ──
@dsl.pipeline(name="iris-e2e", description="data -> train -> eval -> gate -> register")
def iris_pipeline(
    model_bucket: str = "models",
    version: str = "v1",
    threshold: float = 0.9,
    s3_endpoint: str = "",       # MinIO면 http://minio-service.kubeflow:9000 등
):
    data = prepare_data()

    tr = train(train_data=data.outputs["train_out"])

    ev = evaluate(
        model_in=tr.outputs["model_out"],
        test_data=data.outputs["test_out"],
    )

    # 품질 게이트: accuracy >= threshold 일 때만 등록 스텝 실행
    with dsl.If(ev.output >= threshold, name="quality-gate"):
        reg = register_model(
            model_in=tr.outputs["model_out"],
            bucket=model_bucket,
            version=version,
            s3_endpoint=s3_endpoint,
        )
        # S3 자격증명(secret s3-creds) → 등록 스텝의 env 로 주입
        kubernetes.use_secret_as_env(
            reg,
            secret_name="s3-creds",
            secret_key_to_env={
                "AWS_ACCESS_KEY_ID": "AWS_ACCESS_KEY_ID",
                "AWS_SECRET_ACCESS_KEY": "AWS_SECRET_ACCESS_KEY",
            },
        )
```

> 구버전 SDK는 `dsl.If` 대신 `dsl.Condition(ev.output >= threshold)` 사용.

## run_pipeline.py (컴파일 + 제출)
```python
from kfp import compiler
from kfp.client import Client
from pipeline import iris_pipeline

# (a) 컴파일 → YAML (KFP UI 업로드용으로도 사용 가능)
compiler.Compiler().compile(iris_pipeline, "iris_pipeline.yaml")
print("compiled -> iris_pipeline.yaml")

# (b) 바로 제출 (UI를 port-forward 중이라고 가정: localhost:8080)
client = Client(host="http://localhost:8080")
run = client.create_run_from_pipeline_func(
    iris_pipeline,
    arguments={
        "model_bucket": "models",
        "version": "v1",
        "threshold": 0.9,
        "s3_endpoint": "",       # MinIO 사용 시 엔드포인트 지정
    },
    experiment_name="iris-e2e",
)
print("run id:", run.run_id)
```

실행:
```bash
# KFP UI 포트포워드 (다른 터미널에서 유지)
kubectl port-forward -n kubeflow svc/ml-pipeline-ui 8080:80

# 파이프라인 제출
python run_pipeline.py
```
> 또는 iris_pipeline.yaml 을 KFP UI(localhost:8080) → Pipelines → Upload → Create run 으로 실행.

## 꼭 알아둘 포인트 (gotchas)
| 항목 | 내용 |
|---|---|
| 인터넷 필요 | lightweight 컴포넌트는 런타임에 pip install → 파드가 PyPI 접근 가능해야 함. 에어갭이면 커스텀 이미지 필요 |
| secret 네임스페이스 | s3-creds는 파이프라인이 도는 kubeflow 네임스페이스에 있어야 함 (KServe용은 서빙 ns에 별도) |
| 품질 게이트 동작 | accuracy < threshold면 register_model 스킵 → 아무것도 승격 안 됨(서빙은 기존 버전 유지) |
| 계보/캐싱 | 각 스텝 입출력이 아티팩트로 추적됨. 입력 안 바뀌면 재실행 시 캐시로 skip |
| 모델 파일명 | model.joblib 고정 — 다음 단계 KServe sklearn 런타임이 이 이름을 찾음 |
